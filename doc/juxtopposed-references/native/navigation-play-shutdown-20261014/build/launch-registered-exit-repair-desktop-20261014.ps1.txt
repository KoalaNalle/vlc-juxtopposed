$ErrorActionPreference = 'Stop'
$workspacePath = '/checkout'
$packagePath = Join-Path $workspacePath 'build-windows\slice-music-playlist-package-20261014'
$evidencePath = Join-Path $workspacePath 'build-windows\registered-exit-repair-desktop-20261014'
$profilePath = Join-Path $evidencePath 'userdata'
if (Test-Path -LiteralPath $evidencePath) { throw 'Use a fresh evidence directory' }
if (!(Test-Path -LiteralPath (Join-Path $packagePath 'vlc.exe'))) { throw 'Stage the runtime before creating an evidence directory' }
[System.IO.Directory]::CreateDirectory($profilePath) | Out-Null
foreach ($fixtureDirectory in @('ml','juxtopposed-playlist-art')) {
    $sourcePath = Join-Path (Join-Path $workspacePath 'build-windows\leaf-literal-desktop-20261014\userdata') $fixtureDirectory
    Copy-Item -LiteralPath $sourcePath -Destination (Join-Path $profilePath $fixtureDirectory) -Recurse
}
$startInfo = [System.Diagnostics.ProcessStartInfo]::new()
$startInfo.FileName = Join-Path $packagePath 'vlc.exe'
$startInfo.UseShellExecute = $false
$startInfo.WorkingDirectory = $packagePath
$startInfo.WindowStyle = [System.Diagnostics.ProcessWindowStyle]::Normal
$startInfo.Environment.Remove('VLC_PLUGIN_PATH') | Out-Null
$startInfo.Environment['VLC_LIB_PATH'] = Join-Path $packagePath 'plugins-registered-exit-repair-20261014'
$startInfo.Environment['VLC_LIBEXEC_PATH'] = Join-Path $packagePath 'vlc-preparser.exe'
$startInfo.Environment['VLC_USERDATA_PATH'] = $profilePath
foreach ($argument in @('--qt-juxtopposed-controls', '--no-one-instance', '--qt-pin-controls', '--start-paused', '--no-ml-network-discovery', '--file-logging', "--logfile=$evidencePath\vlc.log", '-vv')) { $startInfo.ArgumentList.Add($argument) }
$startedPlayer = [System.Diagnostics.Process]::Start($startInfo)
Write-Output "Repaired navigation test player PID: $($startedPlayer.Id)"
